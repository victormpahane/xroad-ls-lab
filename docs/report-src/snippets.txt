### p3-script
#!/usr/bin/env bash
# ss-setup.sh <container> <member_code> <server_code> "<O name>"
# Phase 2 steps 1-7 for one Security Server, run from the laptop.
set -u
SS=$1 MC=$2 SC=$3 O=$4
S=./work/$SS; mkdir -p $S
K=$(docker exec $SS curl -sk -u xrd:secret -X POST https://localhost:4000/api/v1/api-keys -H 'Content-Type: application/json' \
   -d '["XROAD_SYSTEM_ADMINISTRATOR","XROAD_SECURITY_OFFICER","XROAD_REGISTRATION_OFFICER","XROAD_SERVICE_ADMINISTRATOR"]' \
   | python3 -c 'import sys,json;print(json.load(sys.stdin)["key"])')
api(){ docker exec -i $SS curl -sk -H "Authorization: X-Road-ApiKey token=$K" "$@"; }
A=https://localhost:4000/api/v1
docker cp ~/Downloads/configuration_anchor_LS-DEV.xml $SS:/tmp/anchor.xml
echo "== [$SS] anchor";  api -X POST $A/system/anchor -H 'Content-Type: application/octet-stream' --data-binary @/tmp/anchor.xml -o /dev/null -w "[HTTP %{http_code}]\n"
echo "== [$SS] init $MC/$SC"; api -X POST $A/initialization -H 'Content-Type: application/json' -d "{\"owner_member_class\":\"GOV\",\"owner_member_code\":\"$MC\",\"security_server_code\":\"$SC\",\"software_token_pin\":\"123456xrd!\",\"ignore_warnings\":true}" -w "[HTTP %{http_code}]\n"
echo "== [$SS] login";  api -X PUT $A/tokens/0/login -H 'Content-Type: application/json' -d '{"password":"123456xrd!"}' -o /dev/null -w "[HTTP %{http_code}]\n"
echo "== [$SS] TSA";    api -X POST $A/system/timestamping-services -H 'Content-Type: application/json' -d '{"name":"Test TSA","url":"http://ca:8899"}' -o /dev/null -w "[HTTP %{http_code}]\n"
for u in sign auth; do
  if [ $u = sign ]; then US=SIGNING; X=",\"member_id\":\"LS-DEV:GOV:$MC\""; F="{\"O\":\"$O\",\"subjectAltName\":\"$SS\"}"
  else US=AUTHENTICATION; X=""; F="{\"O\":\"$O\",\"CN\":\"$SS\",\"subjectAltName\":\"$SS\"}"; fi
  r=$(api -X POST $A/tokens/0/keys-with-csrs -H 'Content-Type: application/json' -d "{\"key_label\":\"$SS-$u\",\"csr_generate_request\":{\"key_usage_type\":\"$US\",\"ca_name\":\"Test CA\",\"csr_format\":\"DER\"$X,\"subject_field_values\":$F}}")
  read KID CID < <(echo "$r" | python3 -c 'import sys,json;d=json.load(sys.stdin);print(d["key"]["id"],d["csr_id"])') || { echo "CSR FAIL: $r"; exit 1; }
  api "$A/keys/$KID/csrs/$CID?csr_format=DER" > $S/${SS}_$u.der
  curl -s -F certreq=@$S/${SS}_$u.der -F type=$u -o $S/${SS}_$u.pem http://localhost:8888/testca/sign
  echo "== [$SS] $u cert: $(openssl x509 -in $S/${SS}_$u.pem -noout -subject)"
done
H=$(api -X POST $A/token-certificates -H 'Content-Type: application/octet-stream' --data-binary @- < $S/${SS}_auth.pem | python3 -c 'import sys,json;print(json.load(sys.stdin)["certificate_details"]["hash"])')
echo "== [$SS] activate auth"; api -X PUT $A/token-certificates/$H/activate -o /dev/null -w "[HTTP %{http_code}]\n"
echo "== [$SS] import sign"; api -X POST $A/token-certificates -H 'Content-Type: application/octet-stream' --data-binary @- < $S/${SS}_sign.pem -o /dev/null -w "[HTTP %{http_code}]\n"
echo "== [$SS] register auth at $SS (via ss1 management services)"; api -X PUT $A/token-certificates/$H/register -H 'Content-Type: application/json' -d "{\"address\":\"$SS\"}" -w "[HTTP %{http_code}]\n"
